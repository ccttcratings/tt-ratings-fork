class Player {

    this.won_against = [];

    this.scores_l = {};

}

function sortWithIndeces(toSort, order) {
  for (var i = 0; i < toSort.length; i++) {
    toSort[i] = [toSort[i], i];
  }
  if (order == "ascending") {
    toSort.sort(function(left, right) {
      return left[0] < right[0] ? -1 : 1;
    });
  } else {
    toSort.sort(function(left, right) {
      return left[0] > right[0] ? -1 : 1;
    });
  }
  toSort.sortIndices = [];
  for (var j = 0; j < toSort.length; j++) {
    toSort.sortIndices.push(toSort[j][1]);
    toSort[j] = toSort[j][0];
  }
  return toSort;
}

function tiebreaker(player_list) {
  // for (var i = 0; i < player_list.length; ++i) {
  //   Logger.log(player_list[i].name + "  " + player_list[i].won_games);
  // }
  if (player_list.length == 2) {
    Logger.log("Two way tiebreak.")
    if (player_list[0].won_against.includes(player_list[1].name)) {
      return [player_list[0], player_list[1]];
    } else if (player_list[1].won_against.includes(player_list[0].name)) {
      return [player_list[1], player_list[0]];
    } else {
      Logger.log("check scores");
      var p1_w_total = player_list[0].scores_w[player_list[1].name].reduce((a, b) => a + b, 0);
      var p1_l_total = player_list[0].scores_l[player_list[1].name].reduce((a, b) => a + b, 0);
      var p2_w_total = player_list[1].scores_w[player_list[0].name].reduce((a, b) => a + b, 0);
      var p2_l_total = player_list[1].scores_l[player_list[0].name].reduce((a, b) => a + b, 0);
      var p1_check = p1_w_total / p1_l_total;
      var p2_check = p2_w_total / p2_l_total;
      Logger.log(player_list[0].name + " = " + p1_w_total + " / " + p1_l_total + " = " + p1_check);
      Logger.log(player_list[1].name + " = " + p2_w_total + " / " + p2_l_total + " = " + p2_check);
      if (p1_check > p2_check) {
        return [player_list[0], player_list[1]];
      } else {
        return [player_list[1], player_list[0]];
      }
    }
  } else if (player_list.length == 3) {
    Logger.log("Three way tiebreak.")
    
    if (player_list[0].won_against.includes(player_list[1].name)) {
      // 0 won 1, 1 won 2, 2 won 0
      var p1 = player_list[0];
      var p2 = player_list[1];
      var p3 = player_list[2];
      Logger.log("0 won 1, 1 won 2, 2 won 0");
    } else if (player_list[0].won_against.includes(player_list[2].name)) {
      // 0 won 2, 2 won 1, 1 won 0
      var p1 = player_list[0];
      var p2 = player_list[2];
      var p3 = player_list[1];
      Logger.log("0 won 2, 2 won 1, 1 won 0");
    }

    var new_player_list = [p1, p2, p3];
    var ret_player_list = [];
    if (p1.match_diffs[p2.name].length == p2.match_diffs[p3.name].length && 
        p2.match_diffs[p3.name].length == p3.match_diffs[p1.name].length) {
      var score_diffs = [p1.match_diffs[p2.name].reduce((a, b) => a + b, 0), 
                         p2.match_diffs[p3.name].reduce((a, b) => a + b, 0),
                         p3.match_diffs[p1.name].reduce((a, b) => a + b, 0)];
      score_diffs = sortWithIndeces(score_diffs, "decending");
      for (var i = 0; i < 3; ++i) {
        var index = score_diffs.sortIndices[i];
        ret_player_list.push(new_player_list[index]);
      }
    } else {
      var game_counts = [p1.match_diffs[p2.name].length, p2.match_diffs[p3.name].length, p3.match_diffs[p1.name].length];
      game_counts = sortWithIndeces(game_counts, "ascending");
      if (game_counts[0] == game_counts[1]) {
        var i_1 = game_counts.sortIndices[0];
        var i_2 = game_counts.sortIndices[1];
        var i_3 = game_counts.sortIndices[2];
        if (new_player_list[i_1].won_against.includes(new_player_list[i_2].name)) {
          ret_player_list = [new_player_list[i_1], new_player_list[i_2], new_player_list[i_3]];
        } else {
          ret_player_list = [new_player_list[i_2], new_player_list[i_1], new_player_list[i_3]];
        }
      } else {
        for (var i = 0; i < 3; ++i) {
          var index = game_counts.sortIndices[i];
          ret_player_list.push(new_player_list[index]);
        }
      }
    }
    return ret_player_list;
  }
}

function findMatchWinner(scores) {
  var p1_win_count = 0;
  var p2_win_count = 0;
  var p1_scores_w = [];
  var p1_scores_l = [];
  var p2_scores_w = [];
  var p2_scores_l = [];
  var p1vp2_score_diffs = [];
  var p2vp1_score_diffs = [];

  // Debug: Log the scores array
  // Logger.log("  findMatchWinner - scores: " + scores.join(","));

  for (var i = 0; i < scores.length / 2; ++i) {
    var p1_col = i * 2;
    var p2_col = p1_col + 1;

    // Make sure we have both scores
    if (p2_col >= scores.length) {
      break;
    }

    var p1_score = scores[p1_col];
    var p2_score = scores[p2_col];

    // Skip if either score is not a number
    if (typeof p1_score !== 'number' || typeof p2_score !== 'number') {
      continue;
    }

    var score_diff = p1_score - p2_score;
    if (score_diff > 0) {
      ++p1_win_count;
    } else if (score_diff < 0) {
      ++p2_win_count;
    }
    // If score_diff == 0, it's a tie game (11-11?) which shouldn't happen in table tennis

    p1_scores_w.push(p1_score);
    p1_scores_l.push(p2_score);
    p2_scores_w.push(p2_score);
    p2_scores_l.push(p1_score);
    p1vp2_score_diffs.push(score_diff);
    p2vp1_score_diffs.push(-score_diff);
  }

  // Logger.log("  findMatchWinner - p1_wins: " + p1_win_count + ", p2_wins: " + p2_win_count);

  winner_id = -1;
  if (p1_win_count > p2_win_count) {
    winner_id = 1;
  } else if (p1_win_count < p2_win_count) {
    winner_id = 2;
  } else if (p1_win_count == 0 && p2_win_count == 0) {
    winner_id = 0;  // No games played
  } else {
    winner_id = 3;  // Tie
  }
  return [winner_id, p1vp2_score_diffs, p1_scores_w, p1_scores_l, p2vp1_score_diffs, p2_scores_w, p2_scores_l]
}

function findWinners() {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getActiveSheet();

    // Check if we're on a date sheet (format YYYY-MM-DD)
    var sheetName = sheet.getName();
    if (sheetName === "Ratings" || sheetName === "Template") {
      // Try to find the most recent date sheet
      var sheets = ss.getSheets();
      var datePattern = /^\d{4}-\d{2}-\d{2}$/;
      var dateSheets = [];

      for (var i = 0; i < sheets.length; i++) {
        if (datePattern.test(sheets[i].getName())) {
          dateSheets.push(sheets[i].getName());
        }
      }

      if (dateSheets.length > 0) {
        // Sort and get the most recent date
        dateSheets.sort();
        sheetName = dateSheets[dateSheets.length - 1];
        sheet = ss.getSheetByName(sheetName);
        SpreadsheetApp.setActiveSheet(sheet);
      } else {
        SpreadsheetApp.getUi().alert("Please run this on a date sheet (e.g., 2025-09-27), not on the Ratings sheet.");
        return;
      }
    }

    // Add a visual indicator that the script is running
    SpreadsheetApp.getActiveSpreadsheet().toast('Finding winners for sheet: ' + sheetName, 'Processing', 3);

  var player_list = sheet.getRangeList(["B2:B7", "B19:B24", "B36:B41"]).getRanges();
  var league_results = {};
  for (var i = 0; i < 3; ++i) {
    var results = {};
    var player_names = player_list[i].getValues();
    for (var j = 0; j < 6; j++) {
      if (player_names[j][0]) {
        var name = String(player_names[j][0]).trim();
        if (name !== "") {
          results[name] = new Player(name);
        }
      }
    }
    league_results[i] = results;
    Logger.log("League " + (i+1) + " players: " + Object.keys(results).join(", "));
  }
  
  var score_list = sheet.getRangeList(["H2:T16", "H19:T33", "H36:T50"]).getRanges();
  for (var i = 0; i < 3; ++i) {
    var base_index = i * 17 + 2;
    // Clear background for columns F and G (point/table winner columns)
    sheet.getRange(base_index, 6, 15, 2).setBackground("#d9d9d9");  // Columns F and G

    var league_scores = score_list[i].getValues();
    for (var j = 0; j < 15; ++j) {
      var match_scores = league_scores[j];
      // Now Player 1 is in column H (index 0) and Player 2 is in column J (index 2)
      if (!match_scores[0] || !match_scores[2]) {
        continue;
      }
      // Trim whitespace from player names
      var p1_name = String(match_scores[0]).trim();
      var p2_name = String(match_scores[2]).trim();

      if (p1_name === "" || p2_name === "") {
        continue;
      }

      // Make sure both players exist in the league_results
      if (!league_results[i][p1_name] || !league_results[i][p2_name]) {
        Logger.log("Warning: Player not found in league " + i + ": '" + p1_name + "' or '" + p2_name + "'");
        Logger.log("Available players: " + Object.keys(league_results[i]).join(", "));
        continue;
      }

      // Filter to get only the numeric scores (skipping column I which has ELO change)
      var scores_only = [];
      for (var k = 3; k < match_scores.length; k++) {  // Start from index 3 (column K)
        if (typeof match_scores[k] === "number") {
          scores_only.push(match_scores[k]);
        }
      }

      // Make sure we have scores to process
      if (scores_only.length == 0) {
        continue;
      }

      var [winner_id, p1vp2_score_diffs, p1_scores_w, p1_scores_l, p2vp1_score_diffs, p2_scores_w, p2_scores_l] =
          findMatchWinner(scores_only);

      Logger.log("Match " + (j+1) + ": " + p1_name + " vs " + p2_name + " - Winner ID: " + winner_id + ", Scores: " + scores_only.join(","));

      if (winner_id == 1) {
        ++league_results[i][p1_name].won_games;
        league_results[i][p1_name].won_against.push(p2_name);
        sheet.getRange(base_index + j, 6).setBackground("#ea9999");  // Column F (winner 1)
        Logger.log("  -> " + p1_name + " wins");
      } else if (winner_id == 2) {
        ++league_results[i][p2_name].won_games;
        league_results[i][p2_name].won_against.push(p1_name);
        sheet.getRange(base_index + j, 7).setBackground("#ea9999");  // Column G (winner 2)
        Logger.log("  -> " + p2_name + " wins");
      } else if (winner_id == 3) {
        sheet.getRange(base_index + j, 6).setBackground("#b3a7d7");  // Column F (tie)
        sheet.getRange(base_index + j, 7).setBackground("#b3a7d7");  // Column G (tie)
        Logger.log("  -> Tie");
      } else {
        Logger.log("  -> No winner determined (winner_id = " + winner_id + ")");
      }

      league_results[i][p1_name].match_diffs[p2_name] = p1vp2_score_diffs;
      league_results[i][p2_name].match_diffs[p1_name] = p2vp1_score_diffs;
      league_results[i][p1_name].scores_w[p2_name] = p1_scores_w;
      league_results[i][p1_name].scores_l[p2_name] = p1_scores_l;
      league_results[i][p2_name].scores_w[p1_name] = p2_scores_w;
      league_results[i][p2_name].scores_l[p1_name] = p2_scores_l;
    }
  }

  for (var i = 0; i < 3; i++) {
    var winner = null;
    var player_names = Object.keys(league_results[i]);
    var ties = [];

    // Debug: Log all players and their wins
    Logger.log("League " + (i+1) + " results:");
    for (var j = 0; j < player_names.length; ++j) {
      Logger.log("  " + player_names[j] + ": " + league_results[i][player_names[j]].won_games + " wins");
    }

    // Find the player with most wins
    var max_wins = -1;
    for (var j = 0; j < player_names.length; ++j) {
      var name = player_names[j];
      var wins = league_results[i][name].won_games;

      if (wins > max_wins) {
        max_wins = wins;
        winner = league_results[i][name];
        ties = [];
      } else if (wins == max_wins && max_wins > 0) {
        // We have a tie
        if (ties.length == 0 && winner != null) {
          ties.push(winner);
        }
        ties.push(league_results[i][name]);
      }
    }

    // Handle ties
    if (ties.length > 1) {
      Logger.log("Tie detected with " + ties.length + " players at " + max_wins + " wins");
      var p_list = tiebreaker(ties);
      winner = p_list[0];
    }

    // If no winner found (all have 0 wins), skip
    if (winner == null || winner.won_games == 0) {
      Logger.log("No winner found for league " + (i+1));
      continue;
    }
    var range, ties_range;
    if (i == 0) {
      range = "B10";
      ties_range = "B12:B13";
    } else if (i == 1) {
      range = "B27";
      ties_range = "B29:B30";
    } else {
      range = "B44";
      ties_range = "B46:B47";
    }

    // Write the winner name
    Logger.log("League " + (i+1) + " winner: " + winner.name);
    sheet.getRange(range).setValue(winner.name);

    // Write tie information if there were ties
    if (ties.length > 1) {
      var ties_names = [];
      for (var j = 1; j < 3; ++j) {
        if (j < ties.length) {
          ties_names.push([ties[j].name]);
        } else {
          ties_names.push([""]);
        }
      }
      sheet.getRange(ties_range).setValues(ties_names);
    } else {
      // Clear tie cells if no ties
      sheet.getRange(ties_range).clearContent();
    }
  }

  // Show completion message
  SpreadsheetApp.getActiveSpreadsheet().toast('Winners found and highlighted!', 'Complete', 3);

  } catch (error) {
    Logger.log("Error in findWinners: " + error.toString());
    SpreadsheetApp.getActiveSpreadsheet().toast('Error: ' + error.toString(), 'Error', 10);
    throw error;
  }
}

function sortDictByValue(toSort) {
  var sorted = [];
  for (var i = 0; i < toSort.length; i++) {
    sorted[i] = [toSort[i], i];
  }
  
  sorted.sort(function(left, right) {
    return left[0] > right[0] ? -1 : 1;
  });
  sorted.sortIndices = [];
  for (var j = 0; j < sorted.length; j++) {
    sorted.sortIndices.push(sorted[j][1]);
    sorted[j] = sorted[j][0];
  }
  return sorted;
}
