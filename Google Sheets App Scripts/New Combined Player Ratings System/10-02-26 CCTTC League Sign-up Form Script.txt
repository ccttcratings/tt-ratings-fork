function newLeagueDate() {
  var form = FormApp.openById("14a73b_Qcgus5fV3DmNzSGUFxFcZ8ibqYbux1qxs_bVE");
  var confirmation_msg = "Your response has been recorded. \n\n" +
                         "Registered players: ccttcratings.github.io/portal/ \n\n" +
                         "To sign up or edit your response: tinyurl.com/ccttc-sign-up";
                         
  form.setConfirmationMessage(confirmation_msg);                    
  var now = new Date();
  var ss_id = form.getDestinationId();
  if (!ss_id) { ss_id = '1_Ou_wAKDovm5ASpc85cpGGbxKZcJkiEIEgifOE0ClBk'; }
  var ss = SpreadsheetApp.openById(ss_id);
  var skip_days = 6 - now.getDay();
  skip_days = skip_days < 1 ? 7 : skip_days;
  var next_sat = new Date(now.getTime());
  next_sat.setDate(now.getDate() + skip_days);
  var options = {month: 'short'};
  var new_date_str = `${new Intl.DateTimeFormat('en-US', options).format(next_sat)} ${next_sat.getDate()}, ${next_sat.getFullYear()}`;
  var raw_sheet_name = 'Raw ' + new_date_str;
  form.setAcceptingResponses(true);
  form.setConfirmationMessage(confirmation_msg);
  form.deleteAllResponses();
  var attendanceQuestion = form.getItemById(598333706);
  if (attendanceQuestion) {
    var mcItem = attendanceQuestion.asMultipleChoiceItem();
    mcItem.setChoiceValues(["Yes", "No"]);
  }
  var sheet = ss.getSheetByName(raw_sheet_name);
  if (sheet) {
    Logger.log('Sheet ' + raw_sheet_name + ' already exists — skipping creation.');
  } else {
    try {
      form.removeDestination();
    } catch(e) {
      Logger.log('Form was not linked or could not unlink: ' + e.message);
    }
    SpreadsheetApp.flush();
    form.setDestination(FormApp.DestinationType.SPREADSHEET, ss.getId());
    SpreadsheetApp.flush();
    var allSheets = ss.getSheets();
    var newlyLinkedSheet = null;
    for (var s = 0; s < allSheets.length; s++) {
      var sName = allSheets[s].getName();
      if (sName.indexOf("Form Responses") !== -1 || sName.indexOf("Responses from Form") !== -1) {
        newlyLinkedSheet = allSheets[s];
        break;
      }
    }
    if (!newlyLinkedSheet) {
      newlyLinkedSheet = ss.getSheets();
    }
    newlyLinkedSheet.setName(raw_sheet_name);
    newlyLinkedSheet.hideSheet();
  }
  var currentDescription = form.getDescription();
  var updatedDescription = currentDescription.replace(/\([A-Za-z]{3}\s\d{1,2},\s\d{4}\)/, "(" + new_date_str + ")");
  form.setDescription(updatedDescription);
  var statusPad = "\u00A0\u00A0\u00A0\u00A0\u00A0\u00A0\u00A0\u00A0\u00A0\u00A0\u00A0\u00A0\u00A0\u00A0\u00A0\u00A0\u00A0\u00A0\u00A0";
  var datePad   = "\u00A0\u00A0\u00A0\u00A0\u00A0\u00A0\u00A0\u00A0\u00A0\u00A0\u00A0\u00A0\u00A0\u00A0\u00A0\u00A0\u00A0\u00A0\u00A0\u00A0\u00A0\u00A0\u00A0\u00A0\u00A0\u00A0";
  var boldStatusText = makeTextBoldUnicode("Registration OPEN");
  var boldDateStr    = makeTextBoldUnicode(new_date_str);
  var form_description = `${statusPad}${boldStatusText}\n${datePad}(${boldDateStr})`;
  form.setDescription(form_description);
}

function sendConfirmationEmail(e) {
  try {
    var form = e.source || FormApp.getActiveForm();
    var attendanceQuestionId = 598333706; 
    
    // Check response limits
    limitYesResponses(form, attendanceQuestionId);
    
    var itemResponses = e.response.getItemResponses();
    var playerName = "Player";
    var playerEmail = "";
    var playingStatus = "";
    
    // Extract responses directly by their physical layout slots
    if (itemResponses && itemResponses.length > 0) {
      playerName = String(itemResponses[0].getResponse()).trim();
    }
    if (itemResponses && itemResponses.length > 1) {
      playerEmail = String(itemResponses[1].getResponse()).trim();
    }
    if (itemResponses && itemResponses.length > 2) {
      playingStatus = String(itemResponses[2].getResponse()).trim();
    }
    
    // Ultimate fallback if position data layout shifted
    if (!playerEmail || playerEmail.indexOf('@') === -1) {
      playerEmail = e.response.getRespondentEmail() || "";
    }
    
    // Debug tracer logs
    Logger.log("Captured Name: " + playerName);
    Logger.log("Captured Email: " + playerEmail);
    Logger.log("Captured Status: " + playingStatus);
    
    if (!playerEmail || playerEmail.indexOf('@') === -1) {
      Logger.log("Could not send receipt: No valid email target address could be extracted.");
      return;
    }

    // NEW: Dynamically generates the unique edit link for this specific player submission
    var editResponseUrl = e.response.getEditResponseUrl();
    
    var subject = "League Play Sign-up Confirmation";
    var htmlBody = "<p>Hi " + playerName + ",</p>" +
                   "<p>Your response has been successfully recorded for this Saturday's League Play at 9 a.m. - noon.  Doors open at 7:30 a.m.</p>" +
                   "<p>• Attending: <strong>" + playingStatus + "</strong></p>" +
                   "<p>Registered players: <a href='https://ccttcratings.github.io/portal/'>View players</a></p>" +
                   // NEW: Adds the clickable edit link text right below the player portal link
                   "<p>Need to change your status? <a href='" + editResponseUrl + "'>Edit your response</a></p>" +
                   "<p>See you at the tables!<br>CCTTC Automation System</p>";
                   
    MailApp.sendEmail(playerEmail, subject, "", { 
      htmlBody: htmlBody, 
      name: 'CCTTC Automation System' 
    });
    
    Logger.log("Confirmation receipt successfully sent to: " + playerEmail);
  } catch(error) {
    Logger.log("Error sending player confirmation email: " + error.toString());
  }
}

function limitYesResponses(form, itemId) {
  var item = form.getItemById(itemId);
  if (!item) {
    Logger.log('Item not found. Check the item ID.');
    return;
  }
  
  var responses = form.getResponses();
  var yesCount = 0;
  
  for (var i = 0; i < responses.length; i++) {
    var itemResponses = responses[i].getItemResponses();
    for (var j = 0; j < itemResponses.length; j++) {
      var itemResponse = itemResponses[j];
      if (itemResponse.getItem().getId() === itemId) {
        var answer = itemResponse.getResponse();
        if (answer === "Yes") {
          yesCount++;
        }
      }
    }
  }
  
  Logger.log('Number of active Yes checks: ' + yesCount);
  if (yesCount >= 15) {
    form.setAcceptingResponses(false);
    form.setCustomClosedFormMessage("League registration is full."); 
    Logger.log('Form cap reached. Registration has been closed automatically.');
  }
}

function makeTextBoldUnicode(plainText) {
  var chars = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789";
  var boldCharsArray = [
    "𝗔","𝗕","𝗖","𝗗","𝗘","𝗙","𝗚","𝗛","𝗜","𝗝","𝗞","𝗟","𝗠","𝗡","𝗢","𝗣","𝗤","𝗥","𝗦","𝗧","𝗨","𝗩","𝗪","𝗫","𝗬","𝗭",
    "𝗮","𝗯","𝗰","𝗱","𝗲","𝗳","𝗴","𝗵","𝗶","𝗷","𝗸","𝗹","𝗺","𝗻","𝗼","𝗽","𝗾","𝗿","𝘀","𝘁","𝘂","𝘃","𝘄","𝘅","𝘆","𝘇",
    "𝟬","𝟭","𝟮","𝟯","𝟰","𝟱","𝟲","𝟳","𝟴","𝟵"
  ];
  var result = "";
  for (var i = 0; i < plainText.length; i++) {
    var charIdx = chars.indexOf(plainText[i]);
    if (charIdx !== -1) {
      result += boldCharsArray[charIdx];
    } else {
      result += plainText[i];
    }
  }
  return result;
}
