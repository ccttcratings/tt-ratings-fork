    // =========================================================================
    // 1. SYNC LOCK: Force Google to commit all pseudonym translations first!
    // =========================================================================
    SpreadsheetApp.flush();

    var publicHistorySheet = publicSS.getSheetByName('Ratings History');
    var publicRatingsSheet = publicSS.getSheetByName('🔵 Ratings');

    if (publicHistorySheet && publicRatingsSheet) {
      // Generate a clean tournament date string matching your historical tracking schema
      var currentDate = new Date();
      var displayNameDate = formatDateForSheet(getNextSaturday(currentDate));
      
      // Map out existing public history log entries for today's date to prevent duplicate row rows on re-runs
      var historyLastRow = publicHistorySheet.getLastRow();
      var existingEntries = {};
      
      if (historyLastRow >= 2) {
        var existingData = publicHistorySheet.getRange(1, 1, historyLastRow, 3).getValues();
        for (var r = 1; r < existingData.length; r++) {
          var pName = String(existingData[r][0]).trim();
          var pDateRaw = existingData[r][1];
          if (pName && pDateRaw) {
            var pDateStr = (pDateRaw instanceof Date) ? formatDateForSheet(pDateRaw) : String(pDateRaw).trim();
            if (pDateStr === displayNameDate) {
              existingEntries[pName.toLowerCase()] = r + 1;
            }
          }
        }
      }

      // =========================================================================
      // 2. THE ROSTER SCANPASS: Loop through EVERY player on the polished public sheet
      // =========================================================================
      var ratingsLastRow = publicRatingsSheet.getLastRow();
      if (ratingsLastRow >= 2) {
        // Read Column B (Names), Column C (Ratings), and Column D (Dates) in one quick pass
        var rosterValues = publicRatingsSheet.getRange(2, 2, ratingsLastRow - 1, 3).getValues();
        
        for (var p = 0; p < rosterValues.length; p++) {
          var name = String(rosterValues[p][0]).trim();
          var rating = rosterValues[p][1];
          var dateRaw = rosterValues[p][2];
          
          if (name === "") continue;
          
          // Verify if this row represents an active player who logged matches in today's tracking block
          var playerDateStr = (dateRaw instanceof Date) ? formatDateForSheet(dateRaw) : String(dateRaw).trim();
          if (playerDateStr !== displayNameDate) {
            continue; // Skip non-participating players who didn't play today
          }
          
          // Ensure rating is processed as a pure number, fallback to empty string if hidden/blank
          var activeRating = (typeof rating === 'number' && !isNaN(rating)) ? round2(rating) : "";
          var lookupKey = name.toLowerCase();

          if (existingEntries[lookupKey]) {
            // Overwrite the existing record with the fresh rating values on manual re-syncs
            var targetRow = existingEntries[lookupKey];
            publicHistorySheet.getRange(targetRow, 2).setValue(displayNameDate);
            publicHistorySheet.getRange(targetRow, 3).setValue(activeRating);
          } else {
            // 3. COMPLETE 3-COLUMN APPEND WRITE: Passes Name, Text Date, and Rating cleanly!
            publicHistorySheet.appendRow([name, displayNameDate, activeRating]);
            
            // Format Column B string properties so it locks into text presentation alignment natively
            var newLastRow = publicHistorySheet.getLastRow();
            publicHistorySheet.getRange(newLastRow, 2).setNumberFormat("@");
          }
        }
      }
    }

    SpreadsheetApp.flush();
    Logger.log("✅ SUCCESS: '" + sheetName + "' synchronized flawlessly and history logs updated for all active players!");
  } catch (e) {
    Logger.log("Error publishing public score sheet: " + e.toString());
  }
