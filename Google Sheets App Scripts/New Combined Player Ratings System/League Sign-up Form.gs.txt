function newLeagueDate() {
  var form = FormApp.openById("14a73b_Qcgus5fV3DmNzSGUFxFcZ8ibqYbux1qxs_bVE");
  form.setTitle(form.getTitle().replace("Signup(", "Signup\n(").replace(")**", ")\n**"));

  var confirmation_msg = "Your response has been recorded. \n\nRegistered players: https://tinyurl.com/Registered-CCTTC-Players \n\nTo signup or edit your response: https://tinyurl.com/ccttc-sign-up \n\n";
  form.setConfirmationMessage(confirmation_msg);

  var formItems = form.getItems();
  for (var i = 0; i < formItems.length; i++) {
    var title = formItems[i].getTitle();
    if (/email/i.test(title)) {
      formItems[i].setTitle('Email address');
    } else if (/name/i.test(title)) {
      formItems[i].setTitle('Name');
    }
  }
  form.getItemById(598333706).setTitle('Are you playing?');

  var now = new Date();
  var ss_id = form.getDestinationId();
  if (!ss_id) { ss_id = '1_Ou_wAKDovm5ASpc85cpGGbxKZcJkiEIEgifOE0ClBk'; }
  var ss = SpreadsheetApp.openById(ss_id);

  // Calculate next Saturday
  var skip_days = 6 - now.getDay();
  skip_days = skip_days < 1 ? 7 : skip_days;
  var next_sat = new Date(now.getTime());
  next_sat.setDate(now.getDate() + skip_days);

  var options = {month: 'short'};
  var new_date_str = `${new Intl.DateTimeFormat('en-US', options).format(next_sat)} ${next_sat.getDate()}, ${next_sat.getFullYear()}`;
  var raw_sheet_name = 'Raw ' + new_date_str;
  var new_title = `League Play Sign-up\n(${new_date_str})\n** Registration OPEN **`;

  confirmation_msg = "Your response has been recorded. \n\nRegistered players: https://tinyurl.com/Registered-CCTTC-Players \n\nTo signup or edit your response: https://tinyurl.com/ccttc-sign-up \n\n";

  form.setAcceptingResponses(true);
  form.setConfirmationMessage(confirmation_msg);
  form.setTitle(new_title);
  form.deleteAllResponses();
  form.getItemById(598333706).asMultipleChoiceItem().setChoiceValues(["Yes", "No"]);

  var sheet = ss.getSheetByName(raw_sheet_name);
  if (sheet) {
    Logger.log('Sheet ' + raw_sheet_name + ' already exists — skipping creation.');
    return;
  }
  form.setDestination(FormApp.DestinationType.SPREADSHEET, ss.getId());
  SpreadsheetApp.flush();
  sheet = ss.getSheets()[0];
  sheet.setName(raw_sheet_name);
}
     
function limitYesResponses() {
  var form = FormApp.getActiveForm();
  var itemId = 598333706;
  var item = form.getItemById(itemId);
  if (!item) {
    Logger.log('Item not found. Check the item ID.');
    return;
  }
  var responses = form.getResponses();
  var yesCount = 0;
  responses.forEach(function(response) {
    var itemResponses = response.getItemResponses();
    var itemResponse = itemResponses.find(function(r) {
      return r.getItem().getId() === itemId;
    });
    if (itemResponse) {
      var answers = itemResponse.getResponse();
      Logger.log('Answers: ' + answers);
      if (answers && answers.indexOf('Yes') !== -1) {
        yesCount++;
      }
    }
  });
  Logger.log('Number of Yes checks: ' + yesCount);
  if (yesCount >= 15) {
    form.setAcceptingResponses(false);
    Logger.log('Form has been closed due to Yes responses.');
  }
}

function logItemByIds() {

    Logger.log('"' + items[i].getTitle() + '" has ID: ' + items[i].getId());
  }
}
